" (" + r.source + ")" : ""), r.paymentMode, Number(r.amount), r.note || "Ticked Entry" ]); }); return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); }
